## 1. Products — Public Catalog Reads

`src/routes/products.js` — same shape as [13.2.5](/book/part-04-databases-backend-and-full-stack-development/13-backend-engineering-with-nodejs/13-2-building-apis-with-express/13-2-5-building-a-complete-shopcore-crud-api), trimmed to public reads + admin writes:

```javascript
// GET /api/v1/products           ?category=hardware&q=keyboard&page=1&limit=12
// GET /api/v1/products/:id
// POST /api/v1/products          (admin only)
// PATCH /api/v1/products/:id     (admin only)

import { Router } from 'express';
import { pool } from '../db/pool.js';
import { requireAuth } from '../middleware/requireAuth.js';
import { requireRole } from '../middleware/requireRole.js';
import { validateBody } from '../middleware/validate.js';
import { z } from 'zod';

export const productsRouter = Router();

productsRouter.get('/', async (req, res, next) => {
  try {
    const { category, q } = req.query;
    const page = Math.max(1, Number(req.query.page) || 1);
    const limit = Math.min(50, Math.max(1, Number(req.query.limit) || 12));
    const params = [];
    let where = ' WHERE 1=1';
    if (category) { params.push(category); where += ` AND c.slug = $${params.length}`; }
    if (q)        { params.push(`%${q}%`); where += ` AND (p.title ILIKE $${params.length} OR p.sku ILIKE $${params.length})`; }
    params.push(limit, (page - 1) * limit);
    const { rows } = await pool.query(
      `SELECT p.id, p.sku, p.title, p.price_cents, p.stock_qty, c.name AS category
       FROM products p JOIN categories c ON c.id = p.category_id
       ${where}
       ORDER BY p.id
       LIMIT $${params.length - 1} OFFSET $${params.length}`,
      params
    );
    res.json({ page, limit, products: rows });
  } catch (err) { next(err); }
});

productsRouter.get('/:id', async (req, res, next) => {
  try {
    const { rows } = await pool.query(
      `SELECT p.id, p.sku, p.title, p.price_cents, p.stock_qty, c.name AS category
       FROM products p JOIN categories c ON c.id = p.category_id WHERE p.id = $1`,
      [Number(req.params.id)]
    );
    if (!rows[0]) return res.status(404).json({ error: 'NOT_FOUND' });
    res.json(rows[0]);
  } catch (err) { next(err); }
});

productsRouter.post('/', requireAuth, requireRole('admin'), validateBody(/* productSchema from 13.2.5 */), async (req, res, next) => { /* INSERT ... */ });
productsRouter.patch('/:id', requireAuth, requireRole('admin'), async (req, res, next) => { /* dynamic UPDATE ... */ });
```

## 2. Cart — Server-Side Session Cart

A cart is per-user state: the API owns it (never localStorage) so refresh/device-switch keeps it.

```sql
-- 004_cart.sql
BEGIN;
CREATE TABLE cart_items (
  user_id     BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  product_id  BIGINT NOT NULL REFERENCES products(id) ON DELETE CASCADE,
  qty         INTEGER NOT NULL CHECK (qty > 0),
  PRIMARY KEY (user_id, product_id)
);
INSERT INTO schema_migrations (version) VALUES ('004_cart') ON CONFLICT DO NOTHING;
COMMIT;
```

`src/routes/cart.js` (requires auth — mount under `requireAuth`):

```javascript
// GET    /api/v1/cart                 → items joined with current prices
// PUT    /api/v1/cart/items/:productId { qty }   (qty 0 → remove)
// DELETE /api/v1/cart/items/:productId

cartRouter.put('/items/:productId', validateBody(z.object({ qty: z.number().int().min(0).max(99) })), async (req, res, next) => {
  const productId = Number(req.params.productId);
  const { qty } = req.body;
  if (qty === 0) {
    await pool.query('DELETE FROM cart_items WHERE user_id = $1 AND product_id = $2', [req.user.user_id, productId]);
    return res.status(204).end();
  }
  await pool.query(
    `INSERT INTO cart_items (user_id, product_id, qty) VALUES ($1, $2, $3)
     ON CONFLICT (user_id, product_id) DO UPDATE SET qty = EXCLUDED.qty`,
    [req.user.user_id, productId, qty]
  );
  res.json({ ok: true });
});
```

## 3. Checkout — The Critical Transaction

`POST /api/v1/orders/checkout` — one atomic unit:

```javascript
import { pool } from '../db/pool.js';

checkoutRouter.post('/checkout', async (req, res, next) => {
  const userId = req.user.user_id;
  const client = await pool.connect();            // dedicated client — NOT pool.query
  try {
    await client.query('BEGIN');

    // 1. Read the cart with the products' CURRENT price, locked
    const { rows: items } = await client.query(
      `SELECT ci.product_id, ci.qty, p.price_cents, p.stock_qty, p.title
       FROM cart_items ci JOIN products p ON p.id = ci.product_id
       WHERE ci.user_id = $1
       ORDER BY ci.product_id
       FOR UPDATE OF p`,
      [userId]
    );
    if (items.length === 0) { await client.query('ROLLBACK'); return res.status(400).json({ error: 'CART_EMPTY' }); }

    // 2. Stock check (the atomic guard)
    for (const it of items) {
      if (it.stock_qty < it.qty) {
        await client.query('ROLLBACK');
        return res.status(409).json({ error: 'OUT_OF_STOCK', productId: it.product_id, available: it.stock_qty });
      }
    }

    // 3. Create order + line items (snapshot prices!) + decrement stock
    const total = items.reduce((s, it) => s + it.qty * it.price_cents, 0);
    const { rows: [order] } = await client.query(
      `INSERT INTO orders (user_id, status) VALUES ($1, 'paid') RETURNING id`,
      [userId]
    );
    for (const it of items) {
      await client.query(
        `INSERT INTO order_items (order_id, product_id, qty, unit_price_cents)
         VALUES ($1, $2, $3, $4)`,
        [order.id, it.product_id, it.qty, it.price_cents]   // snapshot!
      );
      await client.query(
        `UPDATE products SET stock_qty = stock_qty - $1
         WHERE id = $2 AND stock_qty >= $1`,
        [it.qty, it.product_id]
      );
    }
    await client.query(
      `INSERT INTO payments (order_id, amount_cents, provider, status)
       VALUES ($1, $2, 'manual', 'succeeded')`,
      [order.id, total]
    );
    await client.query('DELETE FROM cart_items WHERE user_id = $1', [userId]);

    await client.query('COMMIT');
    res.status(201).json({ orderId: order.id, total_cents: total });
  } catch (err) {
    await client.query('ROLLBACK');
    next(err);
  } finally {
    client.release();
  }
});
```

Every guarantee in this block maps to an earlier lesson: snapshot prices (11.1.2/11.1.3), `FOR UPDATE` + `WHERE stock_qty >= n` atomic stock guard (11.2.1), single transaction for multi-table writes (15.1.2), `BIGINT` ids (11.1.1), dedicated client vs pool (13.2.4).

## 4. Verify Checkout — curl Journey

```bash
# Ada logs in (15.2.2), adds KB-01 ×1 to cart
curl -b jar.txt -X PUT http://localhost:4000/api/v1/cart/items/1 \
  -H 'Content-Type: application/json' -d '{"qty": 1}'

# Checkout
curl -b jar.txt -X POST http://localhost:4000/api/v1/orders/checkout
# -> {"orderId":4,"total_cents":12900}

# Stock decremented by 1; cart now empty; order visible to owner only
curl -b jar.txt http://localhost:4000/api/v1/cart          # -> {"items": []}
psql -U postgres -h localhost -d shopcore -c "SELECT id, status FROM orders ORDER BY id;"
```

Also prove the guard: set `products.stock_qty = 0` for KB-01 temporarily, checkout again with KB-01 in cart → expect `409 OUT_OF_STOCK`, and no partial writes (orders table unchanged, cart still intact — because the whole transaction rolled back).

## 5. Practice

**Guided exercise.** Implement products + cart + orders as above. Reproduce the full checkout journey in curl, including the `OUT_OF_STOCK` rollback path.

**Independent exercise.** Add `GET /api/v1/orders` — list the caller's own orders with items (JOIN, `WHERE user_id = $1`). Empty for a fresh customer; after checkout, 1 order with 2 items.

**Challenge.** Add idempotent checkout: require an `Idempotency-Key` header; store it in a `checkout_keys` table; duplicate requests return the original order instead of creating a second one. — **Implemented and verified in the reference backend:** `examples/shopcore/db/migrations/005_checkout_keys.sql` + `src/routes/orders.js` replay the stored order for a repeated key (`200` + same `orderId`, `idempotentReplay: true`) instead of charging twice; the integration suite proves stock is not decremented twice. (13.4.1 patterns.)

**Assessment checklist**

- [ ] Catalog endpoints are public; writes require auth + role.
- [ ] Cart state lives server-side per user.
- [ ] Checkout is one transaction — all-or-nothing on stock, order, items, payment, cart wipe.
- [ ] The `OUT_OF_STOCK` path leaves zero residue (verified in psql).
- [ ] I can price a multi-item order with integer cents and no floats.
