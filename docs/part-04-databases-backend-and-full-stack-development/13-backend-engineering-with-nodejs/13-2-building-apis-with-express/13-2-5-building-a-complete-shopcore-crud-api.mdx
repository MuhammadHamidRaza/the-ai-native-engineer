## 2. Errors and Validation (Condensed from 13.2.3)

**`src/errors.js`:**

```javascript
export class AppError extends Error {
  constructor(statusCode, code, message) {
    super(message);
    this.statusCode = statusCode;
    this.code = code;
  }
}

export function errorBoundary(err, req, res, _next) {
  if (err instanceof AppError) {
    return res.status(err.statusCode).json({ error: err.code, message: err.message });
  }
  console.error(err);
  res.status(500).json({ error: 'INTERNAL_ERROR', message: 'Something went wrong' });
}
```

**`src/middleware/validate.js`:**

```javascript
import { z } from 'zod';

export function validateBody(schema) {
  return (req, res, next) => {
    const parsed = schema.safeParse(req.body);
    if (!parsed.success) {
      return res.status(400).json({
        error: 'VALIDATION_ERROR',
        issues: parsed.error.issues.map((i) => ({ path: i.path.join('.'), message: i.message })),
      });
    }
    req.body = parsed.data;
    next();
  };
}
```

## 3. The Products Router — `src/routes/products.js`

```javascript
import { Router } from 'express';
import { z } from 'zod';
import { pool } from '../db/pool.js';
import { AppError } from '../errors.js';
import { validateBody } from '../middleware/validate.js';

export const productsRouter = Router();

const productSchema = z.object({
  category_id: z.number().int().positive(),
  sku: z.string().min(2).max(40).regex(/^[A-Z0-9-]+$/, 'SKU must be uppercase letters, digits and hyphens'),
  title: z.string().min(2).max(200),
  price_cents: z.number().int().nonnegative(),
  stock_qty: z.number().int().nonnegative().default(0),
});
const productPatchSchema = productSchema.partial().strict();   // all fields optional, no unknown keys

// LIST
productsRouter.get('/', async (_req, res, next) => {
  try {
    const { rows } = await pool.query(
      `SELECT p.id, p.sku, p.title, p.price_cents, p.stock_qty, c.name AS category
       FROM products p JOIN categories c ON c.id = p.category_id
       ORDER BY p.id`
    );
    res.json(rows);
  } catch (err) { next(err); }
});

// GET ONE
productsRouter.get('/:id', async (req, res, next) => {
  try {
    const id = Number(req.params.id);
    if (!Number.isInteger(id) || id <= 0) throw new AppError(400, 'BAD_ID', 'id must be a positive integer');
    const { rows } = await pool.query('SELECT * FROM products WHERE id = $1', [id]);
    if (rows.length === 0) throw new AppError(404, 'NOT_FOUND', `product ${id} does not exist`);
    res.json(rows[0]);
  } catch (err) { next(err); }
});

// CREATE
productsRouter.post('/', validateBody(productSchema), async (req, res, next) => {
  try {
    const { category_id, sku, title, price_cents, stock_qty } = req.body;
    const { rows } = await pool.query(
      `INSERT INTO products (category_id, sku, title, price_cents, stock_qty)
       VALUES ($1, $2, $3, $4, $5) RETURNING *`,
      [category_id, sku, title, price_cents, stock_qty]
    );
    res.status(201).json(rows[0]);
  } catch (err) {
    if (err.code === '23505') return next(new AppError(409, 'SKU_TAKEN', 'sku already exists'));
    if (err.code === '23503') return next(new AppError(400, 'BAD_CATEGORY', 'category_id does not exist'));
    next(err);
  }
});

// UPDATE (partial)
productsRouter.patch('/:id', validateBody(productPatchSchema), async (req, res, next) => {
  try {
    const id = Number(req.params.id);
    if (!Number.isInteger(id) || id <= 0) throw new AppError(400, 'BAD_ID', 'id must be a positive integer');
    const allowed = ['category_id', 'sku', 'title', 'price_cents', 'stock_qty'];
    const fields = Object.entries(req.body).filter(([k]) => allowed.includes(k));
    if (fields.length === 0) throw new AppError(400, 'NO_FIELDS', `provide at least one of: ${allowed.join(', ')}`);
    const sets = fields.map(([k], i) => `${k} = $${i + 1}`).join(', ');
    const values = fields.map(([, v]) => v);
    const { rows } = await pool.query(
      `UPDATE products SET ${sets} WHERE id = $${fields.length + 1} RETURNING *`,
      [...values, id]
    );
    if (rows.length === 0) throw new AppError(404, 'NOT_FOUND', `product ${id} does not exist`);
    res.json(rows[0]);
  } catch (err) { next(err); }
});

// DELETE
productsRouter.delete('/:id', async (req, res, next) => {
  try {
    const id = Number(req.params.id);
    if (!Number.isInteger(id) || id <= 0) throw new AppError(400, 'BAD_ID', 'id must be a positive integer');
    const { rowCount } = await pool.query('DELETE FROM products WHERE id = $1', [id]);
    if (rowCount === 0) throw new AppError(404, 'NOT_FOUND', `product ${id} does not exist`);
    res.status(204).end();
  } catch (err) {
    if (err.code === '23503') return next(new AppError(409, 'PRODUCT_IN_ORDERS', 'product appears in orders; cannot delete'));
    next(err);
  }
});
```

`err.code` values are Postgres **SQLSTATE** codes: `23505` unique violation, `23503` FK violation — always translate them into your API's error vocabulary.

## 4. Wire Everything — `src/app.js`

```javascript
import express from 'express';
import { productsRouter } from './routes/products.js';
import { errorBoundary } from './errors.js';

export const app = express();

app.use(express.json());

app.get('/health', (_req, res) => res.json({ status: 'ok' }));

app.use('/api/v1/products', productsRouter);

// 404 for anything unmatched — AFTER routes, BEFORE boundary
app.use((_req, res) => res.status(404).json({ error: 'NOT_FOUND', message: 'route does not exist' }));

// Error boundary — MUST be last
app.use(errorBoundary);
```

**`src/server.js`:** as 13.2.4, with the Postgres ping before listen:

```javascript
import { app } from './app.js';
import { pool } from './db/pool.js';

await pool.query('SELECT 1');          // fail fast if DB is unreachable
const PORT = process.env.PORT || 4000;
app.listen(PORT, () => console.log(`shopcore-api on http://localhost:${PORT}`));
```

## 5. Run and Verify — curl, With Expected Output

```bash
npm run dev

# list the seeded products
curl http://localhost:4000/api/v1/products
# [{"id":1,"sku":"KB-01","title":"Mechanical Keyboard","price_cents":12900,"stock_qty":15,"category":"Hardware"}, ...]

# create a product
curl -X POST http://localhost:4000/api/v1/products \
  -H "Content-Type: application/json" \
  -d '{"category_id":2,"sku":"BK-04","title":"The Pragmatic Programmer","price_cents":5500,"stock_qty":12}'
# -> 201 {"id":4,"sku":"BK-04",...}

# duplicate SKU -> 409
curl -X POST http://localhost:4000/api/v1/products \
  -H "Content-Type: application/json" \
  -d '{"category_id":1,"sku":"BK-04","title":"Duplicate","price_cents":1}'
# -> {"error":"SKU_TAKEN","message":"sku already exists"}

# partial update -> 200
curl -X PATCH http://localhost:4000/api/v1/products/4 \
  -H "Content-Type: application/json" -d '{"stock_qty": 10}'

# delete -> 204, then GET -> 404
curl -X DELETE http://localhost:4000/api/v1/products/4 -i | head -n 1
curl http://localhost:4000/api/v1/products/4
```

If every output matches, your ShopCore backend is genuinely working — not a demo, a system.

## 6. Troubleshooting

| Symptom | Fix |
| :-- | :-- |
| `Cannot use import statement` | Files must be ESM — `"type": "module"` and correct extensions everywhere |
| `relation "products" does not exist` | Wrong database — connect to `shopcore` (check `DATABASE_URL`), or you never ran the seed |
| `23503` on POST | `category_id` you sent doesn't exist — seed categories (ids start at 1) |
| PATCH 400 "provide at least one of..." | Empty/typo'd field names in body — exact names only | the manual `allowed` list plus `productPatchSchema` (strict) reject unknown keys; drop one for a `VALIDATION_ERROR` |
| 500 on DELETE of referenced product | Expected 409 via state code mapping — if you see 500, the SQLSTATE branch didn't match; log the error |
| Body arrives empty | Missing/incorrect `Content-Type: application/json` header |
